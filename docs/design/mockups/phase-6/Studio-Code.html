1// Postal code validation per country. Raw input in, verdict out.
2export type Country = 'CA' | 'US' | 'GB' | 'NL'
3
4export interface PostalVerdict {
5  ok: boolean
6  numeric: boolean      // drives inputMode on the field
7  message?: string      // plain English, with an example
8}
9
10const RULES: Record<Country, { re: RegExp; numeric: boolean; example: string }> = {
11  CA: { re: /^[A-Z]\d[A-Z] ?\d[A-Z]\d$/i, numeric: false, example: 'K1P 5G1' },
12  US: { re: /^\d{5}(-\d{4})?$/, numeric: true, example: '94107' },
13  GB: { re: /^(GIR 0AA|[A-Z]{1,2}\d[A-Z\d]? ?\d[A-Z]{2})$/i, numeric: false, example: 'SW1A 1AA' },
14  NL: { re: /^\d{4} ?[A-Z]{2}$/i, numeric: false, example: '1012 AB' },
15}
16
17export function validatePostal(country: Country, raw: string): PostalVerdict {
18  const rule = RULES[country]
19  const code = raw.trim().replace(/\s+/g, ' ')   // autocomplete formats; we validate raw
20  if (rule.re.test(code)) return { ok: true, numeric: rule.numeric }
21  return {
22    ok: false,
23    numeric: rule.numeric,
24    message: `Postal codes for ${countryName(country)} look like ${rule.example}`,
25  }
26}
27
28function countryName(c: Country): string {
29  return { CA: 'Canada', US: 'the US', GB: 'the UK', NL: 'the Netherlands' }[c]
30}
